# .htaccess für das Verzeichnis mailer-php
#
# Beim Hochladen als ".htaccess" in den Ordner mailer-php legen.

# Suchmaschinen fernhalten. Der Zugriffsschutz liegt nicht hier, sondern im
# Anmeldelink — dieser Header verhindert nur die Indexierung.
<IfModule mod_headers.c>
  Header always set X-Robots-Tag "noindex, nofollow, noarchive"
</IfModule>

# Verzeichnisinhalte nicht auflisten
Options -Indexes

# Nur index.php ist von aussen erreichbar. Die eingebundenen Dateien
# enthalten den Brevo-Schlüssel und dürfen nie direkt aufgerufen werden.
<FilesMatch "^(konfig|konfig\.beispiel|brevo)\.php$">
  Require all denied
</FilesMatch>

DirectoryIndex index.php
